    /**
     * Обробник події (Listener) зміни значення `input[type="file"]` для завантаження та парсингу Excel-файлів.
     * Зчитує файл, конфігурує SheetJS, валідує структуру аркушів та імпортує дані в IndexedDB через контролер.
     * 
     * @param {Event} ev - Стандартна подія addEventListener від елемента інпуту файлу.
     * @returns {void} Функція нічого не повертає, оскільки працює асинхронно черезFileReader.
     */
    xlsxLoader(ev) {
        // Отримуємо дескриптор вибраного користувачем файлу
        const file = ev.target.files[0];
        if (!file) return;
        
        const reader = new FileReader();

        // 1. Налаштовуємо подію, яка спрацює ПІСЛЯ зчитування файлу
        reader.onload = (ev) => {
            // Перетворюємо сирий буфер результату в сумісний масив байтів (Uint8Array)
            const data = new Uint8Array(ev.target.result);

            // 2. Головна магія SheetJS: парсимо файл
            let readOptions = { // параметри для xlsx
                type: undefined,        //undefined
                raw: false,             //false
                dense: false,           //false 
                codepage: undefined,    //undefined
                cellFormula: true,      //true
                cellHTML: true,         //true
                cellNF: false,          //false
                cellStyles: false,      //false
                cellText: true,         //true
                cellDates: true,       //false
                dateNF: undefined,      //undefined
                sheetStubs: false,      //false
                sheetRows: 0,           //0
                bookDeps: false,        //false
                bookFiles: false,       //false
                bookProps: false,       //false
                bookSheets: false,      //false
                bookVBA: false,         //false
                password: "",           //""
                WTF: false,             //false
                sheets: undefined,      //undefined
                nodim: false,           //false
                PRN: false,             //false
                xlfn: false,            //false
                FS: undefined,          //undefined
                UTC: true               //true
            };
            const workbook = XLSX.read(data, readOptions);
            console.log("workbook: ", workbook);
            
            const storeModel = {};
            let validVersion = { flag: false, diff: [] };
            let tables = {};
            for (let i = 0; i < workbook.SheetNames.length; i++) {
                // працюємо тільки з дозволеними таблицями
                this.model.xlsSheets.includes(workbook.SheetNames[i])
                    // отримуємо масив ключів, задаємо адреси коміркам.
                    && (storeModel[workbook.SheetNames[i]] = this.model.getStorageModel(workbook, workbook.SheetNames[i]))
                    // перевіряємо наявність колекції ключів та їх відповідність з отриманими
                    && (validVersion = this.model.checkVersion(workbook.SheetNames[i], storeModel[workbook.SheetNames[i]]))
                    // формуємо дані для внесення в базу даних через dexie
                    && (tables[workbook.SheetNames[i]] = this.model.getDBTable(workbook.SheetNames[i], workbook.Sheets[workbook.SheetNames[i]], storeModel[workbook.SheetNames[i]]));
            }

            // залити/оновити базу даних
            console.log("controllers: xlsx: storeModel: ", storeModel);
            for (let table in tables) {
                this.model.pushToDB(table, tables[table]);
            }

            

            
        };

        // 2. Запускаємо саме зчитування файлу як ArrayBuffer
        reader.readAsArrayBuffer(file);

        
    }
